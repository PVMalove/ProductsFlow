+import { useInfiniteQuery, QueryKey } from '@tanstack/react-query';
+import { OffsetApiResponse } from '@/lib/api/users';
+
+export function getOffsetQueryOptions<T>(
+  queryKey: QueryKey,
+  fetchFn: (pageIndex: number) => Promise<OffsetApiResponse<T[]>>
+) {
+  return {
+    queryKey,
+    queryFn: ({ pageParam = 1 }: { pageParam: number }) => fetchFn(pageParam),
+    initialPageParam: 1,
+    getNextPageParam: (lastPage: OffsetApiResponse<T[]>) => {
+      const { page_index, total_pages } = lastPage.meta;
+      if (page_index < total_pages) {
+        return page_index + 1;
+      }
+      return null;
+    },
+  };
+}
+
+export function useOffsetInfiniteQuery<T>(
+  queryKey: QueryKey,
+  fetchFn: (pageIndex: number) => Promise<OffsetApiResponse<T[]>>
+) {
+  return useInfiniteQuery(getOffsetQueryOptions<T>(queryKey, fetchFn));
+}

+import { apiClient } from '../apiClient';
+
+export interface UserAuditEntry {
+  id: number;
+  user_id: string;
+  actor_user_id: string;
+  action: string;
+  description: string;
+  created_at: string;
+}
+
+export interface OffsetPageMeta {
+  page_index: number;
+  page_size: number;
+  total: number;
+  total_pages: number;
+}
+
+export interface OffsetApiResponse<T> {
+  data: T;
+  meta: OffsetPageMeta;
+}
+
+export async function getGlobalAuditLog(pageIndex: number = 1, pageSize: number = 20): Promise<OffsetApiResponse<UserAuditEntry[]>> {
+  const res = await apiClient.get<OffsetApiResponse<UserAuditEntry[]>>('/v1/users/audit', {
+    params: {
+      page_index: pageIndex,
+      page_size: pageSize,
+    },
+  });
+  return res.data;
+}
